8/// A unsync storage. This is the default storage type.
9#[derive(Default)]
10pub struct UnsyncStorage(RefCell<Option<Box<dyn std::any::Any>>>);

12impl<T: 'static> Storage<T> for UnsyncStorage {
13    fn try_read(
14        &'static self,
15        #[cfg(any(debug_assertions, feature = "debug_ownership"))]
16        at: crate::GenerationalRefBorrowInfo,
17    ) -> Result<Self::Ref<'static, T>, error::BorrowError> {
18        let borrow = self.0.try_borrow();
19
20        #[cfg(any(debug_assertions, feature = "debug_ownership"))]
21        let borrow = borrow.map_err(|_| at.borrowed_from.borrow_error())?;
22
23        #[cfg(not(any(debug_assertions, feature = "debug_ownership")))]
24        let borrow = borrow.map_err(|_| {
25            error::BorrowError::AlreadyBorrowedMut(error::AlreadyBorrowedMutError {})
26        })?;
27
28        match Ref::filter_map(borrow, |any| any.as_ref()?.downcast_ref()) {
29            Ok(guard) => Ok(GenerationalRef::new(
30                guard,
31                #[cfg(any(debug_assertions, feature = "debug_ownership"))]
32                at,
33            )),
34            Err(_) => Err(error::BorrowError::Dropped(error::ValueDroppedError {
35                #[cfg(any(debug_assertions, feature = "debug_ownership"))]
36                created_at: at.created_at,
37            })),
38        }
39    }
40
41    fn try_write(
42        &'static self,
43        #[cfg(any(debug_assertions, feature = "debug_ownership"))]
44        at: crate::GenerationalRefMutBorrowInfo,
45    ) -> Result<Self::Mut<'static, T>, error::BorrowMutError> {
46        let borrow = self.0.try_borrow_mut();
47
48        #[cfg(any(debug_assertions, feature = "debug_ownership"))]
49        let borrow = borrow.map_err(|_| at.borrowed_from.borrow_mut_error())?;
50
51        #[cfg(not(any(debug_assertions, feature = "debug_ownership")))]
52        let borrow = borrow
53            .map_err(|_| error::BorrowMutError::AlreadyBorrowed(error::AlreadyBorrowedError {}))?;
54
55        RefMut::filter_map(borrow, |any| any.as_mut()?.downcast_mut())
56            .map_err(|_| {
57                error::BorrowMutError::Dropped(error::ValueDroppedError {
58                    #[cfg(any(debug_assertions, feature = "debug_ownership"))]
59                    created_at: at.created_at,
60                })
61            })
62            .map(|guard| {
63                GenerationalRefMut::new(
64                    guard,
65                    #[cfg(any(debug_assertions, feature = "debug_ownership"))]
66                    at,
67                )
68            })
69    }
70
71    fn set(&self, value: T) {
72        *self.0.borrow_mut() = Some(Box::new(value));
73    }
74
75    fn take(&'static self) -> Option<T> {
76        self.0
77            .borrow_mut()
78            .take()
79            .map(|any| *any.downcast().unwrap())
80    }
81}

83thread_local! {
84    static UNSYNC_RUNTIME: RefCell<Vec<MemoryLocation<UnsyncStorage>>> = const { RefCell::new(Vec::new()) };
85}

87impl AnyStorage for UnsyncStorage {
88    type Ref<'a, R: ?Sized + 'static> = GenerationalRef<Ref<'a, R>>;
89    type Mut<'a, W: ?Sized + 'static> = GenerationalRefMut<RefMut<'a, W>>;
90
91    fn downcast_lifetime_ref<'a: 'b, 'b, T: ?Sized + 'static>(
92        ref_: Self::Ref<'a, T>,
93    ) -> Self::Ref<'b, T> {
94        ref_
95    }
96
97    fn downcast_lifetime_mut<'a: 'b, 'b, T: ?Sized + 'static>(
98        mut_: Self::Mut<'a, T>,
99    ) -> Self::Mut<'b, T> {
100        mut_
101    }
102
103    fn try_map<I: ?Sized + 'static, U: ?Sized + 'static>(
104        _self: Self::Ref<'_, I>,
105        f: impl FnOnce(&I) -> Option<&U>,
106    ) -> Option<Self::Ref<'_, U>> {
107        let GenerationalRef {
108            inner,
109            #[cfg(any(debug_assertions, feature = "debug_borrows"))]
110            borrow,
111            ..
112        } = _self;
113        Ref::filter_map(inner, f).ok().map(|inner| GenerationalRef {
114            inner,
115            #[cfg(any(debug_assertions, feature = "debug_borrows"))]
116            borrow,
117        })
118    }
119
120    fn try_map_mut<I: ?Sized + 'static, U: ?Sized + 'static>(
121        mut_ref: Self::Mut<'_, I>,
122        f: impl FnOnce(&mut I) -> Option<&mut U>,
123    ) -> Option<Self::Mut<'_, U>> {
124        let GenerationalRefMut {
125            inner,
126            #[cfg(any(debug_assertions, feature = "debug_borrows"))]
127            borrow,
128            ..
129        } = mut_ref;
130        RefMut::filter_map(inner, f)
131            .ok()
132            .map(|inner| GenerationalRefMut {
133                inner,
134                #[cfg(any(debug_assertions, feature = "debug_borrows"))]
135                borrow,
136            })
137    }
138
139    fn data_ptr(&self) -> *const () {
140        self.0.as_ptr() as *const ()
141    }
142
143    fn manually_drop(&self) -> bool {
144        self.0.borrow_mut().take().is_some()
145    }
146
147    fn claim() -> MemoryLocation<Self> {
148        UNSYNC_RUNTIME.with(|runtime| {
149            if let Some(location) = runtime.borrow_mut().pop() {
150                location
151            } else {
152                let data: &'static MemoryLocationInner =
153                    &*Box::leak(Box::new(MemoryLocationInner {
154                        data: Self::default(),
155                        #[cfg(any(debug_assertions, feature = "check_generation"))]
156                        generation: 0.into(),
157                        #[cfg(any(debug_assertions, feature = "debug_borrows"))]
158                        borrow: Default::default(),
159                    }));
160                MemoryLocation(data)
161            }
162        })
163    }
164
165    fn recycle(location: &MemoryLocation<Self>) {
166        let location = *location;
167        location.drop();
168        UNSYNC_RUNTIME.with(|runtime| runtime.borrow_mut().push(location));
169    }
170}
