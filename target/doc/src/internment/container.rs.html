6pub struct TypeHolderSend(Vec<AnySend>);
7
8struct AnySend(Box<dyn Any + Send>);
9
10impl TypeHolderSend {
11    pub fn get_type_mut<T: Any + Send + Default>(&mut self) -> &mut T {
12        if let Some(i) = self
13            .0
14            .iter_mut()
15            .position(|x| x.0.downcast_mut::<T>().is_some())
16        {
17            self.0[i].0.downcast_mut().unwrap()
18        } else {
19            let v: T = Default::default();
20            self.0.push(AnySend(Box::new(v)));
21            self.0.last_mut().unwrap().0.downcast_mut().unwrap()
22        }
23    }
24    pub const fn new() -> Self {
25        TypeHolderSend(Vec::new())
26    }
27}

29const INTERN_CONTAINER_COUNT: usize = 32;
30pub struct Arena {
31    containers: [Mutex<TypeHolderSend>; INTERN_CONTAINER_COUNT],
32}
33
34impl Arena {
35    pub const fn new() -> Self {
36        const EMPTY: Mutex<TypeHolderSend> = parking_lot::const_mutex(TypeHolderSend::new());
37        Arena {
38            containers: [EMPTY; INTERN_CONTAINER_COUNT],
39        }
40    }
41
42    pub fn with<F, T, R>(&self, f: F) -> R
43    where
44        F: FnOnce(&mut T) -> R,
45        T: Any + Send + Default,
46    {
47        // Compute the hash of the type.
48        fn hash_of_type<T: 'static>() -> u64 {
49            // We use very simple hasher, because it is optimized away to a constant:
50            // https://rust.godbolt.org/z/4T1fa4GGs
51            // which is not true for using `DefaultHasher`:
52            // https://rust.godbolt.org/z/qKar1WKfz
53            struct HasherForTypeId {
54                hash: u64,
55            }
56
57            impl Hasher for HasherForTypeId {
58                fn write(&mut self, bytes: &[u8]) {
59                    // Hash for type only calls `write_u64` once,
60                    // but handle this case explicitly to make sure
61                    // this code doesn't break if stdlib internals change.
62
63                    for byte in bytes {
64                        self.hash = self.hash.wrapping_mul(31).wrapping_add(*byte as u64);
65                    }
66                }
67
68                fn write_u64(&mut self, v: u64) {
69                    self.hash = v;
70                }
71
72                fn finish(&self) -> u64 {
73                    self.hash
74                }
75            }
76
77            let mut hasher = HasherForTypeId { hash: 0 };
78            TypeId::of::<T>().hash(&mut hasher);
79            hasher.hash
80        }
81
82        f(
83            self.containers[hash_of_type::<T>() as usize % INTERN_CONTAINER_COUNT]
84                .lock()
85                .get_type_mut(),
86        )
87    }
88}
