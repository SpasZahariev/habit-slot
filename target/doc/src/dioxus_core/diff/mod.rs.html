17    pub(crate) fn create_children<'a>(
18        &mut self,
19        to: &mut impl WriteMutations,
20        nodes: impl IntoIterator<Item = &'a VNode>,
21        parent: Option<ElementRef>,
22    ) -> usize {
23        nodes
24            .into_iter()
25            .map(|child| child.create(self, to, parent))
26            .sum()
27    }

29    /// Simply replace a placeholder with a list of nodes
30    fn replace_placeholder<'a>(
31        &mut self,
32        to: &mut impl WriteMutations,
33        placeholder_id: ElementId,
34        r: impl IntoIterator<Item = &'a VNode>,
35        parent: Option<ElementRef>,
36    ) {
37        let m = self.create_children(to, r, parent);
38        to.replace_node_with(placeholder_id, m);
39        self.reclaim(placeholder_id);
40    }

42    fn nodes_to_placeholder(
43        &mut self,
44        to: &mut impl WriteMutations,
45        mount: MountId,
46        dyn_node_idx: usize,
47        old_nodes: &[VNode],
48    ) {
49        // Create the placeholder first, ensuring we get a dedicated ID for the placeholder
50        let placeholder = self.next_element();
51
52        // Set the id of the placeholder
53        self.mounts[mount.0].mounted_dynamic_nodes[dyn_node_idx] = placeholder.0;
54
55        to.create_placeholder(placeholder);
56
57        self.replace_nodes(to, old_nodes, 1);
58    }

60    /// Replace many nodes with a number of nodes on the stack
61    fn replace_nodes(&mut self, to: &mut impl WriteMutations, nodes: &[VNode], m: usize) {
62        debug_assert!(
63            !nodes.is_empty(),
64            "replace_nodes must have at least one node"
65        );
66
67        // We want to optimize the replace case to use one less mutation if possible
68        // Instead of *just* removing it, we can use the replace mutation
69        self.remove_nodes(to, nodes, Some(m));
70    }

72    /// Remove these nodes from the dom
73    /// Wont generate mutations for the inner nodes
74    fn remove_nodes(
75        &mut self,
76        to: &mut impl WriteMutations,
77        nodes: &[VNode],
78        replace_with: Option<usize>,
79    ) {
80        for (i, node) in nodes.iter().rev().enumerate() {
81            let last_node = i == nodes.len() - 1;
82            node.remove_node(self, to, replace_with.filter(|_| last_node), true);
83        }
84    }

86    pub(crate) fn remove_component_node(
87        &mut self,
88        to: &mut impl WriteMutations,
89        scope: ScopeId,
90        replace_with: Option<usize>,
91        gen_muts: bool,
92    ) {
93        // Remove the component from the dom
94        if let Some(node) = self.scopes[scope.0].last_rendered_node.take() {
95            node.remove_node(self, to, replace_with, gen_muts)
96        };
97
98        // Now drop all the resources
99        self.drop_scope(scope);
100    }

102    /// Insert a new template into the VirtualDom's template registry
103    // used in conditional compilation
104    #[allow(unused_mut)]
105    pub(crate) fn register_template(
106        &mut self,
107        to: &mut impl WriteMutations,
108        mut template: Template,
109    ) {
110        let (path, byte_index) = template.name.rsplit_once(':').unwrap();
111
112        let byte_index = byte_index.parse::<usize>().unwrap();
113        // First, check if we've already seen this template
114        if self
115            .templates
116            .get(&path)
117            .filter(|set| set.contains_key(&byte_index))
118            .is_none()
119        {
120            // if hot reloading is enabled, then we need to check for a template that has overriten this one
121            #[cfg(debug_assertions)]
122            if let Some(mut new_template) = self
123                .templates
124                .get_mut(path)
125                .and_then(|map| map.remove(&usize::MAX))
126            {
127                // the byte index of the hot reloaded template could be different
128                new_template.name = template.name;
129                template = new_template;
130            }
131
132            self.templates
133                .entry(path)
134                .or_default()
135                .insert(byte_index, template);
136
137            // If it's all dynamic nodes, then we don't need to register it
138            if !template.is_completely_dynamic() {
139                to.register_template(template)
140            }
141        }
142    }
143
144    /// Insert a new template into the VirtualDom's template registry
145    pub(crate) fn register_template_first_byte_index(&mut self, mut template: Template) {
146        // First, make sure we mark the template as seen, regardless if we process it
147        let (path, _) = template.name.rsplit_once(':').unwrap();
148        if let Some((_, old_template)) = self
149            .templates
150            .entry(path)
151            .or_default()
152            .iter_mut()
153            .min_by_key(|(byte_index, _)| **byte_index)
154        {
155            // the byte index of the hot reloaded template could be different
156            template.name = old_template.name;
157            *old_template = template;
158        } else {
159            // This is a template without any current instances
160            self.templates
161                .entry(path)
162                .or_default()
163                .insert(usize::MAX, template);
164        }
165
166        // If it's all dynamic nodes, then we don't need to register it
167        if !template.is_completely_dynamic() {
168            self.queued_templates.push(template);
169        }
170    }
171}

173/// We can apply various optimizations to dynamic nodes that are the single child of their parent.
174///
175/// IE
176///  - for text - we can use SetTextContent
177///  - for clearing children we can use RemoveChildren
178///  - for appending children we can use AppendChildren
179#[allow(dead_code)]
180fn is_dyn_node_only_child(node: &VNode, idx: usize) -> bool {
181    let template = node.template.get();
182    let path = template.node_paths[idx];
183
184    // use a loop to index every static node's children until the path has run out
185    // only break if the last path index is a dynamic node
186    let mut static_node = &template.roots[path[0] as usize];
187
188    for i in 1..path.len() - 1 {
189        match static_node {
190            TemplateNode::Element { children, .. } => static_node = &children[path[i] as usize],
191            _ => return false,
192        }
193    }
194
195    match static_node {
196        TemplateNode::Element { children, .. } => children.len() == 1,
197        _ => false,
198    }
