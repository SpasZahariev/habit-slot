12impl VirtualDom {
13    pub(crate) fn diff_scope(
14        &mut self,
15        to: &mut impl WriteMutations,
16        scope: ScopeId,
17        new_nodes: RenderReturn,
18    ) {
19        self.runtime.scope_stack.borrow_mut().push(scope);
20        let scope_state = &mut self.scopes[scope.0];
21        // Load the old and new bump arenas
22        let new = &new_nodes;
23        let old = scope_state.last_rendered_node.take().unwrap();
24
25        old.diff_node(new, self, to);
26
27        let scope_state = &mut self.scopes[scope.0];
28        scope_state.last_rendered_node = Some(new_nodes);
29
30        self.runtime.scope_stack.borrow_mut().pop();
31    }
32
33    /// Create a new template [`VNode`] and write it to the [`Mutations`] buffer.
34    ///
35    /// This method pushes the ScopeID to the internal scopestack and returns the number of nodes created.
36    pub(crate) fn create_scope(
37        &mut self,
38        to: &mut impl WriteMutations,
39        scope: ScopeId,
40        new_node: RenderReturn,
41        parent: Option<ElementRef>,
42    ) -> usize {
43        self.runtime.scope_stack.borrow_mut().push(scope);
44
45        // Create the node
46        let nodes = new_node.create(self, to, parent);
47
48        // Then set the new node as the last rendered node
49        self.scopes[scope.0].last_rendered_node = Some(new_node);
50
51        self.runtime.scope_stack.borrow_mut().pop();
52        nodes
53    }
54}

56impl VNode {
57    pub(crate) fn diff_vcomponent(
58        &self,
59        mount: MountId,
60        idx: usize,
61        new: &VComponent,
62        old: &VComponent,
63        scope_id: ScopeId,
64        parent: Option<ElementRef>,
65        dom: &mut VirtualDom,
66        to: &mut impl WriteMutations,
67    ) {
68        // Replace components that have different render fns
69        if old.render_fn != new.render_fn {
70            return self.replace_vcomponent(mount, idx, new, parent, dom, to);
71        }
72
73        // copy out the box for both
74        let old_scope = &mut dom.scopes[scope_id.0];
75        let old_props: &mut dyn AnyProps = old_scope.props.deref_mut();
76        let new_props: &dyn AnyProps = new.props.deref();
77
78        // If the props are static, then we try to memoize by setting the new with the old
79        // The target ScopeState still has the reference to the old props, so there's no need to update anything
80        // This also implicitly drops the new props since they're not used
81        if old_props.memoize(new_props.props()) {
82            tracing::trace!("Memoized props for component {:#?}", scope_id,);
83            return;
84        }
85
86        // Now run the component and diff it
87        let new = dom.run_scope(scope_id);
88        dom.diff_scope(to, scope_id, new);
89
90        let height = dom.runtime.get_state(scope_id).unwrap().height;
91        dom.dirty_scopes.remove(&ScopeOrder::new(height, scope_id));
92    }
93
94    fn replace_vcomponent(
95        &self,
96        mount: MountId,
97        idx: usize,
98        new: &VComponent,
99        parent: Option<ElementRef>,
100        dom: &mut VirtualDom,
101        to: &mut impl WriteMutations,
102    ) {
103        let scope = ScopeId(dom.mounts[mount.0].mounted_dynamic_nodes[idx]);
104
105        let m = self.create_component_node(mount, idx, new, parent, dom, to);
106
107        // Instead of *just* removing it, we can use the replace mutation
108        dom.remove_component_node(to, scope, Some(m), true);
109    }
110
111    pub(super) fn create_component_node(
112        &self,
113        mount: MountId,
114        idx: usize,
115        component: &VComponent,
116        parent: Option<ElementRef>,
117        dom: &mut VirtualDom,
118        to: &mut impl WriteMutations,
119    ) -> usize {
120        // Load up a ScopeId for this vcomponent. If it's already mounted, then we can just use that
121        let scope = dom
122            .new_scope(component.props.duplicate(), component.name)
123            .state()
124            .id;
125
126        // Store the scope id for the next render
127        dom.mounts[mount.0].mounted_dynamic_nodes[idx] = scope.0;
128
129        let new = dom.run_scope(scope);
130
131        dom.create_scope(to, scope, new, parent)
132    }
