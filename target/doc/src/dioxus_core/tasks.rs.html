13/// A task's unique identifier.
14///
15/// `Task` is a unique identifier for a task that has been spawned onto the runtime. It can be used to cancel the task
16#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
17#[derive(Copy, Clone, PartialEq, Eq, Hash, Debug)]
18pub struct Task(pub(crate) slotmap::DefaultKey);

20impl Task {
21    /// Start a new future on the same thread as the rest of the VirtualDom.
22    ///
23    /// This future will not contribute to suspense resolving, so you should primarily use this for reacting to changes
24    /// and long running tasks.
25    ///
26    /// Whenever the component that owns this future is dropped, the future will be dropped as well.
27    ///
28    /// Spawning a future onto the root scope will cause it to be dropped when the root component is dropped - which
29    /// will only occur when the VirtualDom itself has been dropped.
30    pub fn new(task: impl Future<Output = ()> + 'static) -> Self {
31        spawn(task)
32    }
33
34    /// Drop the task immediately.
35    ///
36    /// This does not abort the task, so you'll want to wrap it in an abort handle if that's important to you
37    pub fn cancel(self) {
38        remove_future(self);
39    }
40
41    /// Pause the task.
42    pub fn pause(&self) {
43        self.set_active(false);
44    }
45
46    /// Resume the task.
47    pub fn resume(&self) {
48        self.set_active(true);
49    }
50
51    /// Check if the task is paused.
52    pub fn paused(&self) -> bool {
53        Runtime::with(|rt| {
54            if let Some(task) = rt.tasks.borrow().get(self.0) {
55                !task.active.get()
56            } else {
57                false
58            }
59        })
60        .unwrap_or_default()
61    }
62
63    /// Wake the task.
64    pub fn wake(&self) {
65        Runtime::with(|rt| _ = rt.sender.unbounded_send(SchedulerMsg::TaskNotified(*self)));
66    }
67
68    /// Poll the task immediately.
69    pub fn poll_now(&self) -> Poll<()> {
70        Runtime::with(|rt| rt.handle_task_wakeup(*self)).unwrap()
71    }
72
73    /// Set the task as active or paused.
74    pub fn set_active(&self, active: bool) {
75        Runtime::with(|rt| {
76            if let Some(task) = rt.tasks.borrow().get(self.0) {
77                let was_active = task.active.replace(active);
78                if !was_active && active {
79                    _ = rt.sender.unbounded_send(SchedulerMsg::TaskNotified(*self));
80                }
81            }
82        });
83    }
84}

87    /// Start a new future on the same thread as the rest of the VirtualDom.
88    ///
89    /// **You should generally use `spawn` instead of this method unless you specifically need to need to run a task during suspense**
90    ///
91    /// This future will not contribute to suspense resolving but it will run during suspense.
92    ///
93    /// Because this future runs during suspense, you need to be careful to work with hydration. It is not recommended to do any async IO work in this future, as it can easily cause hydration issues. However, you can use isomorphic tasks to do work that can be consistently replicated on the server and client like logging or responding to state changes.
94    ///
95    /// ```rust, no_run
96    /// # use dioxus::prelude::*;
97    /// // ❌ Do not do requests in isomorphic tasks. It may resolve at a different time on the server and client, causing hydration issues.
98    /// let mut state = use_signal(|| None);
99    /// spawn_isomorphic(async move {
100    ///     state.set(Some(reqwest::get("https://api.example.com").await));
101    /// });
102    ///
103    /// // ✅ You may wait for a signal to change and then log it
104    /// let mut state = use_signal(|| 0);
105    /// spawn_isomorphic(async move {
106    ///     loop {
107    ///         tokio::time::sleep(std::time::Duration::from_secs(1)).await;
108    ///         println!("State is {state}");
109    ///     }
110    /// });
111    /// ```
112    pub fn spawn_isomorphic(
113        &self,
114        scope: ScopeId,
115        task: impl Future<Output = ()> + 'static,
116    ) -> Task {
117        self.spawn_task_of_type(scope, task, TaskType::Isomorphic)
118    }

120    /// Start a new future on the same thread as the rest of the VirtualDom.
121    ///
122    /// This future will not contribute to suspense resolving, so you should primarily use this for reacting to changes
123    /// and long running tasks.
124    ///
125    /// Whenever the component that owns this future is dropped, the future will be dropped as well.
126    ///
127    /// Spawning a future onto the root scope will cause it to be dropped when the root component is dropped - which
128    /// will only occur when the VirtualDom itself has been dropped.
129    pub fn spawn(&self, scope: ScopeId, task: impl Future<Output = ()> + 'static) -> Task {
130        self.spawn_task_of_type(scope, task, TaskType::ClientOnly)
131    }
132
133    fn spawn_task_of_type(
134        &self,
135        scope: ScopeId,
136        task: impl Future<Output = ()> + 'static,
137        ty: TaskType,
138    ) -> Task {
139        // Insert the task, temporarily holding a borrow on the tasks map
140        let (task, task_id) = {
141            let mut tasks = self.tasks.borrow_mut();
142
143            let mut task_id = Task(DefaultKey::default());
144            let mut local_task = None;
145            tasks.insert_with_key(|key| {
146                task_id = Task(key);
147
148                let new_task = Rc::new(LocalTask {
149                    scope,
150                    active: Cell::new(true),
151                    parent: self.current_task(),
152                    task: RefCell::new(Box::pin(task)),
153                    waker: futures_util::task::waker(Arc::new(LocalTaskHandle {
154                        id: task_id,
155                        tx: self.sender.clone(),
156                    })),
157                    ty: Cell::new(ty),
158                });
159
160                local_task = Some(new_task.clone());
161
162                new_task
163            });
164
165            (local_task.unwrap(), task_id)
166        };
167
168        // Get a borrow on the task, holding no borrows on the tasks map
169        debug_assert!(self.tasks.try_borrow_mut().is_ok());
170        debug_assert!(task.task.try_borrow_mut().is_ok());
171
172        self.sender
173            .unbounded_send(SchedulerMsg::TaskNotified(task_id))
174            .expect("Scheduler should exist");
175
176        task_id
177    }

179    /// Queue an effect to run after the next render
180    pub(crate) fn queue_effect(&self, id: ScopeId, f: impl FnOnce() + 'static) {
181        // Add the effect to the queue of effects to run after the next render for the given scope
182        let mut effects = self.pending_effects.borrow_mut();
183        let scope_order = ScopeOrder::new(id.height(), id);
184        match effects.get(&scope_order) {
185            Some(effects) => effects.push_back(Box::new(f)),
186            None => {
187                effects.insert(Effect::new(scope_order, f));
188            }
189        }
190    }

206    pub(crate) fn handle_task_wakeup(&self, id: Task) -> Poll<()> {
207        debug_assert!(Runtime::current().is_some(), "Must be in a dioxus runtime");
208
209        let task = self.tasks.borrow().get(id.0).cloned();
210
211        // The task was removed from the scheduler, so we can just ignore it
212        let Some(task) = task else {
213            return Poll::Ready(());
214        };
215
216        // If a task woke up but is paused, we can just ignore it
217        if !task.active.get() {
218            return Poll::Pending;
219        }
220
221        let mut cx = std::task::Context::from_waker(&task.waker);
222
223        // update the scope stack
224        self.scope_stack.borrow_mut().push(task.scope);
225        self.rendering.set(false);
226        self.current_task.set(Some(id));
227
228        let poll_result = task.task.borrow_mut().as_mut().poll(&mut cx);
229
230        if poll_result.is_ready() {
231            // Remove it from the scope so we dont try to double drop it when the scope dropes
232            self.get_state(task.scope)
233                .unwrap()
234                .spawned_tasks
235                .borrow_mut()
236                .remove(&id);
237
238            self.remove_task(id);
239        }
240
241        // Remove the scope from the stack
242        self.scope_stack.borrow_mut().pop();
243        self.rendering.set(true);
244        self.current_task.set(None);
245
246        poll_result
247    }
248
249    /// Drop the future with the given TaskId
250    ///
251    /// This does not abort the task, so you'll want to wrap it in an abort handle if that's important to you
252    pub(crate) fn remove_task(&self, id: Task) -> Option<Rc<LocalTask>> {
253        let task = self.tasks.borrow_mut().remove(id.0);
254        if let Some(task) = &task {
255            if task.suspended() {
256                self.suspended_tasks.set(self.suspended_tasks.get() - 1);
257            }
258        }
259        task
260    }
261
262    /// Check if a task should be run during suspense
263    pub(crate) fn task_runs_during_suspense(&self, task: Task) -> bool {
264        let borrow = self.tasks.borrow();
265        let task: Option<&LocalTask> = borrow.get(task.0).map(|t| &**t);
266        matches!(task, Some(LocalTask { ty, .. }) if ty.get().runs_during_suspense())
267    }

270/// the task itself is the waker
271pub(crate) struct LocalTask {
272    scope: ScopeId,
273    parent: Option<Task>,
274    task: RefCell<Pin<Box<dyn Future<Output = ()> + 'static>>>,
275    waker: Waker,
276    ty: Cell<TaskType>,

280impl LocalTask {
281    pub(crate) fn suspend(&self) {
282        self.ty.set(TaskType::Suspended);
283    }

286        matches!(self.ty.get(), TaskType::Suspended)

303/// The type of message that can be sent to the scheduler.
304///
305/// These messages control how the scheduler will process updates to the UI.
306pub(crate) enum SchedulerMsg {
307    /// Immediate updates from Components that mark them as dirty
308    Immediate(ScopeId),
309
310    /// A task has woken and needs to be progressed
311    TaskNotified(Task),
312
313    /// An effect has been queued to run after the next render
314    EffectQueued,
315}
316
317struct LocalTaskHandle {
318    id: Task,
319    tx: futures_channel::mpsc::UnboundedSender<SchedulerMsg>,
320}
321
322impl ArcWake for LocalTaskHandle {
323    fn wake_by_ref(arc_self: &Arc<Self>) {
324        _ = arc_self
325            .tx
326            .unbounded_send(SchedulerMsg::TaskNotified(arc_self.id));
327    }
