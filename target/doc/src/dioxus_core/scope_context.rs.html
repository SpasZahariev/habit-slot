75    ///
76    /// ## Notice: you should prefer using [`Self::schedule_update_any`] and [`Self::scope_id`]
77    pub fn schedule_update(&self) -> Arc<dyn Fn() + Send + Sync + 'static> {
78        let (chan, id) = (self.sender(), self.id);

103    /// Try to retrieve a shared state with type `T` from any parent scope.
104    ///
105    /// Clones the state if it exists.
106    pub fn consume_context<T: 'static + Clone>(&self) -> Option<T> {
107        tracing::trace!(
108            "looking for context {} ({:?}) in {}",
109            std::any::type_name::<T>(),
110            std::any::TypeId::of::<T>(),
111            self.name
112        );
113        if let Some(this_ctx) = self.has_context() {
114            return Some(this_ctx);
115        }
116
117        let mut search_parent = self.parent_id;
118        let cur_runtime = Runtime::with(|runtime| {
119            while let Some(parent_id) = search_parent {
120                let parent = runtime.get_state(parent_id).unwrap();
121                tracing::trace!(
122                    "looking for context {} ({:?}) in {}",
123                    std::any::type_name::<T>(),
124                    std::any::TypeId::of::<T>(),
125                    parent.name
126                );
127                if let Some(shared) = parent.shared_contexts.borrow().iter().find_map(|any| {
128                    tracing::trace!("found context {:?}", (**any).type_id());
129                    any.downcast_ref::<T>()
130                }) {
131                    return Some(shared.clone());
132                }
133                search_parent = parent.parent_id;
134            }
135            None
136        });
137
138        match cur_runtime.flatten() {
139            Some(ctx) => Some(ctx),
140            None => {
141                tracing::trace!(
142                    "context {} ({:?}) not found",
143                    std::any::type_name::<T>(),
144                    std::any::TypeId::of::<T>()
145                );
146                None
147            }
148        }
149    }

151    /// Inject a Box<dyn Any> into the context of this scope
152    pub(crate) fn provide_any_context(&self, mut value: Box<dyn Any>) {
153        let mut contexts = self.shared_contexts.borrow_mut();
154
155        // If the context exists, swap it out for the new value
156        for ctx in contexts.iter_mut() {
157            // Swap the ptr directly
158            if ctx.as_ref().type_id() == value.as_ref().type_id() {
159                std::mem::swap(ctx, &mut value);
160                return;
161            }
162        }
163
164        // Else, just push it
165        contexts.push(value);
166    }

168    /// Expose state to children further down the [`crate::VirtualDom`] Tree. Requires `Clone` on the context to allow getting values down the tree.
169    ///
170    /// This is a "fundamental" operation and should only be called during initialization of a hook.
171    ///
172    /// For a hook that provides the same functionality, use `use_provide_context` and `use_context` instead.
173    ///
174    /// # Example
175    ///
176    /// ```rust, ignore
177    /// struct SharedState(&'static str);
178    ///
179    /// static app: Component = |cx| {
180    ///     cx.use_hook(|| cx.provide_context(SharedState("world")));
181    ///     rsx!(Child {})
182    /// }
183    ///
184    /// static Child: Component = |cx| {
185    ///     let state = cx.consume_state::<SharedState>();
186    ///     rsx!(div { "hello {state.0}" })
187    /// }
188    /// ```
189    pub fn provide_context<T: 'static + Clone>(&self, value: T) -> T {
190        tracing::trace!(
191            "providing context {} ({:?}) in {}",
192            std::any::type_name::<T>(),
193            std::any::TypeId::of::<T>(),
194            self.name
195        );
196        let mut contexts = self.shared_contexts.borrow_mut();
197
198        // If the context exists, swap it out for the new value
199        for ctx in contexts.iter_mut() {
200            // Swap the ptr directly
201            if let Some(ctx) = ctx.downcast_mut::<T>() {
202                std::mem::swap(ctx, &mut value.clone());
203                return value;
204            }
205        }
206
207        // Else, just push it
208        contexts.push(Box::new(value.clone()));
209
210        value
211    }

213    /// Provide a context to the root and then consume it
214    ///
215    /// This is intended for "global" state management solutions that would rather be implicit for the entire app.
216    /// Things like signal runtimes and routers are examples of "singletons" that would benefit from lazy initialization.
217    ///
218    /// Note that you should be checking if the context existed before trying to provide a new one. Providing a context
219    /// when a context already exists will swap the context out for the new one, which may not be what you want.
220    pub fn provide_root_context<T: 'static + Clone>(&self, context: T) -> T {
221        Runtime::with(|runtime| {
222            runtime
223                .get_state(ScopeId::ROOT)
224                .unwrap()
225                .provide_context(context)
226        })
227        .expect("Runtime to exist")
228    }

230    /// Start a new future on the same thread as the rest of the VirtualDom.
231    ///
232    /// **You should generally use `spawn` instead of this method unless you specifically need to need to run a task during suspense**
233    ///
234    /// This future will not contribute to suspense resolving but it will run during suspense.
235    ///
236    /// Because this future runs during suspense, you need to be careful to work with hydration. It is not recommended to do any async IO work in this future, as it can easily cause hydration issues. However, you can use isomorphic tasks to do work that can be consistently replicated on the server and client like logging or responding to state changes.
237    ///
238    /// ```rust, no_run
239    /// # use dioxus::prelude::*;
240    /// // ❌ Do not do requests in isomorphic tasks. It may resolve at a different time on the server and client, causing hydration issues.
241    /// let mut state = use_signal(|| None);
242    /// spawn_isomorphic(async move {
243    ///     state.set(Some(reqwest::get("https://api.example.com").await));
244    /// });
245    ///
246    /// // ✅ You may wait for a signal to change and then log it
247    /// let mut state = use_signal(|| 0);
248    /// spawn_isomorphic(async move {
249    ///     loop {
250    ///         tokio::time::sleep(std::time::Duration::from_secs(1)).await;
251    ///         println!("State is {state}");
252    ///     }
253    /// });
254    /// ```
255    pub fn spawn_isomorphic(&self, fut: impl Future<Output = ()> + 'static) -> Task {
256        let id = Runtime::with(|rt| rt.spawn_isomorphic(self.id, fut)).expect("Runtime to exist");
257        self.spawned_tasks.borrow_mut().insert(id);
258        id
259    }

279    /// Store a value between renders. The foundational hook for all other hooks.
280    ///
281    /// Accepts an `initializer` closure, which is run on the first use of the hook (typically the initial render).
282    /// `use_hook` will return a clone of the value on every render.
283    ///
284    /// In order to clean up resources you would need to implement the [`Drop`] trait for an inner value stored in a RC or similar (Signals for instance),
285    /// as these only drop their inner value once all references have been dropped, which only happens when the component is dropped.
286    ///
287    /// # Example
288    ///
289    /// ```rust
290    /// use dioxus::prelude::*;
291    ///
292    /// // prints a greeting on the initial render
293    /// pub fn use_hello_world() {
294    ///     use_hook(|| println!("Hello, world!"));
295    /// }
296    /// ```
297    ///
298    /// # Custom Hook Example
299    ///
300    /// ```rust
301    /// use dioxus::prelude::*;
302    ///
303    /// pub struct InnerCustomState(usize);
304    ///
305    /// impl Drop for InnerCustomState {
306    ///     fn drop(&mut self){
307    ///         println!("Component has been dropped.");
308    ///     }
309    /// }
310    ///
311    /// #[derive(Clone, Copy)]
312    /// pub struct CustomState {
313    ///     inner: Signal<InnerCustomState>
314    /// }
315    ///
316    /// pub fn use_custom_state() -> CustomState {
317    ///     use_hook(|| CustomState {
318    ///         inner: Signal::new(InnerCustomState(0))
319    ///     })
320    /// }
321    /// ```
322    pub fn use_hook<State: Clone + 'static>(&self, initializer: impl FnOnce() -> State) -> State {
323        let cur_hook = self.hook_index.get();
324        let mut hooks = self.hooks.try_borrow_mut().expect("The hook list is already borrowed: This error is likely caused by trying to use a hook inside a hook which violates the rules of hooks.");
325
326        if cur_hook >= hooks.len() {
327            hooks.push(Box::new(initializer()));
328        }
329
330        hooks
331            .get(cur_hook)
332            .and_then(|inn| {
333                self.hook_index.set(cur_hook + 1);
334                let raw_ref: &dyn Any = inn.as_ref();
335                raw_ref.downcast_ref::<State>().cloned()
336            })
337            .expect(
338                r#"
339                Unable to retrieve the hook that was initialized at this index.
340                Consult the `rules of hooks` to understand how to use hooks properly.
341
342                You likely used the hook in a conditional. Hooks rely on consistent ordering between renders.
343                Functions prefixed with "use" should never be called conditionally.
344                "#,
345            )
346    }
