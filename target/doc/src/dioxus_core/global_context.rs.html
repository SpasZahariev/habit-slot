59/// Start a new future on the same thread as the rest of the VirtualDom.
60///
61/// **You should generally use `spawn` instead of this method unless you specifically need to run a task during suspense**
62///
63/// This future will not contribute to suspense resolving but it will run during suspense.
64///
65/// Because this future runs during suspense, you need to be careful to work with hydration. It is not recommended to do any async IO work in this future, as it can easily cause hydration issues. However, you can use isomorphic tasks to do work that can be consistently replicated on the server and client like logging or responding to state changes.
66///
67/// ```rust, no_run
68/// # use dioxus::prelude::*;
69/// // ❌ Do not do requests in isomorphic tasks. It may resolve at a different time on the server and client, causing hydration issues.
70/// let mut state = use_signal(|| None);
71/// spawn_isomorphic(async move {
72///     state.set(Some(reqwest::get("https://api.example.com").await));
73/// });
74///
75/// // ✅ You may wait for a signal to change and then log it
76/// let mut state = use_signal(|| 0);
77/// spawn_isomorphic(async move {
78///     loop {
79///         tokio::time::sleep(std::time::Duration::from_secs(1)).await;
80///         println!("State is {state}");
81///     }
82/// });
83/// ```
84pub fn spawn_isomorphic(fut: impl Future<Output = ()> + 'static) -> Task {
85    Runtime::with_current_scope(|cx| cx.spawn_isomorphic(fut)).expect("to be in a dioxus runtime")
86}

114/// Store a value between renders. The foundational hook for all other hooks.
115///
116/// Accepts an `initializer` closure, which is run on the first use of the hook (typically the initial render).
117/// `use_hook` will return a clone of the value on every render.
118///
119/// In order to clean up resources you would need to implement the [`Drop`] trait for an inner value stored in a RC or similar (Signals for instance),
120/// as these only drop their inner value once all references have been dropped, which only happens when the component is dropped.
121///
122/// # Example
123///
124/// ```rust
125/// use dioxus::prelude::*;
126///
127/// // prints a greeting on the initial render
128/// pub fn use_hello_world() {
129///     use_hook(|| println!("Hello, world!"));
130/// }
131/// ```
132///
133/// # Custom Hook Example
134///
135/// ```rust
136/// use dioxus::prelude::*;
137///
138/// pub struct InnerCustomState(usize);
139///
140/// impl Drop for InnerCustomState {
141///     fn drop(&mut self){
142///         println!("Component has been dropped.");
143///     }
144/// }
145///
146/// #[derive(Clone, Copy)]
147/// pub struct CustomState {
148///     inner: Signal<InnerCustomState>
149/// }
150///
151/// pub fn use_custom_state() -> CustomState {
152///     use_hook(|| CustomState {
153///         inner: Signal::new(InnerCustomState(0))
154///     })
155/// }
156/// ```
157pub fn use_hook<State: Clone + 'static>(initializer: impl FnOnce() -> State) -> State {
158    Runtime::with_current_scope(|cx| cx.use_hook(initializer)).expect("to be in a dioxus runtime")
159}

201/// Creates a callback that will be run before the component is removed.
202/// This can be used to clean up side effects from the component
203/// (created with [`use_effect`](crate::use_effect)).
204///
205/// Example:
206/// ```rust, ignore
207/// use dioxus::prelude::*;
208///
209/// fn app() -> Element {
210///     let state = use_signal(|| true);
211///     rsx! {
212///         for _ in 0..100 {
213///             h1 {
214///                 "spacer"
215///             }
216///         }
217///         if **state {
218///             rsx! {
219///                 child_component {}
220///             }
221///         }
222///         button {
223///             onclick: move |_| {
224///                 state.set(!*state.get());
225///             },
226///             "Unmount element"
227///         }
228///     }
229/// }
230///
231/// fn child_component() -> Element {
232///     let original_scroll_position = use_signal(|| 0.0);
233///
234///     use_effect(move |_| async move {
235///         let window = web_sys::window().unwrap();
236///         let document = window.document().unwrap();
237///         let element = document.get_element_by_id("my_element").unwrap();
238///         element.scroll_into_view();
239///         original_scroll_position.set(window.scroll_y().unwrap());
240///     });
241///
242///     use_drop(move || {
243///         /// restore scroll to the top of the page
244///         let window = web_sys::window().unwrap();
245///         window.scroll_with_x_and_y(*original_scroll_position.current(), 0.0);
246///     });
247///
248///     rsx!{
249///         div {
250///             id: "my_element",
251///             "hello"
252///         }
253///     }
254/// }
255/// ```
256pub fn use_drop<D: FnOnce() + 'static>(destroy: D) {
257    struct LifeCycle<D: FnOnce()> {
258        /// Wrap the closure in an option so that we can take it out on drop.
259        ondestroy: Option<D>,
260    }
261
262    /// On drop, we want to run the closure.
263    impl<D: FnOnce()> Drop for LifeCycle<D> {
264        fn drop(&mut self) {
265            if let Some(f) = self.ondestroy.take() {
266                f();
267            }
268        }
269    }
270
271    // We need to impl clone for the lifecycle, but we don't want the drop handler for the closure to be called twice.
272    impl<D: FnOnce()> Clone for LifeCycle<D> {
273        fn clone(&self) -> Self {
274            Self { ondestroy: None }
275        }
276    }
277
278    use_hook(|| LifeCycle {
279        ondestroy: Some(destroy),
280    });
281}

311/// Wait for the next render to complete
312///
313/// This is useful if you've just triggered an update and want to wait for it to finish before proceeding with valid
314/// DOM nodes.
315///
316/// Effects rely on this to ensure that they only run effects after the DOM has been updated. Without wait_for_next_render effects
317/// are run immediately before diffing the DOM, which causes all sorts of out-of-sync weirdness.
318pub async fn wait_for_next_render() {
319    // Wait for the flush lock to be available
320    // We release it immediately, so it's impossible for the lock to be held longer than this function
321    Runtime::with(|rt| rt.render_signal.subscribe())
322        .unwrap()
323        .await;
324}

326/// Use a hook with a cleanup function
327pub fn use_hook_with_cleanup<T: Clone + 'static>(
328    hook: impl FnOnce() -> T,
329    cleanup: impl FnOnce(T) + 'static,
330) -> T {
331    let value = use_hook(hook);
332    let _value = value.clone();
333    use_drop(move || cleanup(_value));
334    value
