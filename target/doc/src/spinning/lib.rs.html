10/// An extremely simple spinlock, locking by compare-and-swapping a single flag and repeating.
11pub struct RawMutex {
12    locked: AtomicBool,
13}
14
15impl RawMutex {
16    pub const fn const_new() -> Self {
17        RawMutex {
18            locked: AtomicBool::new(false),
19        }
20    }
21}

23unsafe impl lock_api::RawMutex for RawMutex {
24    const INIT: Self = RawMutex::const_new();
25
26    type GuardMarker = lock_api::GuardNoSend;
27
28    fn lock(&self) {
29        while !self.try_lock() {
30            atomic::spin_loop_hint();
31        }
32    }
33
34    fn try_lock(&self) -> bool {
35        self.locked
36            .compare_exchange(false, true, Ordering::Acquire, Ordering::Relaxed)
37            .is_ok()
38    }
39    unsafe fn unlock(&self) {
40        let prev = self.locked.fetch_and(false, Ordering::Release);
41        debug_assert!(prev, "unlocking when not locked");
42    }
43}

45#[derive(Debug)]
46pub struct RawRwLock {
47    // The state of the rwlock is composed as follows:
48    //
49    // 1. The base bits, all the way up to ARCH_POINTER_WIDTH - 4, are used to count the number of
50    //    current locks. These also include the one occasional intent or write lock, and the reason
51    //    for this is to allow for more simple atomic operations, since x86 has no instrucion doing
52    //    both bit tests and addition.
53    // 2. RWLOCK_STATE_ACTIVE_WRITER_BIT indicates that the lock currently holds a writer. When
54    //    acquiring a shared lock, the counter may be incremented arbitrarily, but acquiring a
55    //    shared lock must always fail when this bit is set.
56    // 3. RWLOCK_STATE_ACTIVE_INTENT_BIT denotes that the rwlock currently holds an intent lock.
57    //    Attaining a shared lock does ignores this bit, since intent locks only conflict with write
58    //    locks, until they are upgraded to write locks.
59    // 4. RWLOCK_STATE_PENDING_WRITER_BIT can be set at any time by a failed attempt at obtaining a
60    //    write lock, due to shared locks already being present. In order to prevent write lock
61    //    starvation, new shared locks cannot be acquired if this bit is set.
62    state: AtomicUsize,
63}
64impl Clone for RawRwLock {
65    fn clone(&self) -> Self {
66        Self {
67            state: AtomicUsize::new(0),
68        }
69    }
70}
71
72const RWLOCK_STATE_ACTIVE_WRITER_BIT: usize = 1 << (mem::size_of::<usize>() * 8 - 1);
73const RWLOCK_STATE_ACTIVE_INTENT_BIT: usize = 1 << (mem::size_of::<usize>() * 8 - 2);
74
75const RWLOCK_STATE_PENDING_WRITER_BIT: usize = 1 << (mem::size_of::<usize>() * 8 - 3);
76
77const RWLOCK_STATE_EXTRA_MASK: usize = RWLOCK_STATE_ACTIVE_WRITER_BIT
78    | RWLOCK_STATE_ACTIVE_INTENT_BIT
79    | RWLOCK_STATE_PENDING_WRITER_BIT;
80const RWLOCK_STATE_COUNT_MASK: usize = !RWLOCK_STATE_EXTRA_MASK;
81
82impl RawRwLock {
83    pub const fn const_new() -> Self {
84        RawRwLock {
85            state: AtomicUsize::new(0),
86        }
87    }
88
89    fn try_lock_exclusive_raw(&self) -> (bool, bool) {
90        let prev_state = self
91            .state
92            .fetch_or(RWLOCK_STATE_PENDING_WRITER_BIT, Ordering::AcqRel);
93        let current_state = prev_state | RWLOCK_STATE_PENDING_WRITER_BIT;
94        let was_previously_pending = prev_state & RWLOCK_STATE_PENDING_WRITER_BIT != 0;
95
96        if prev_state & RWLOCK_STATE_ACTIVE_INTENT_BIT != 0 {
97            debug_assert_eq!(prev_state & RWLOCK_STATE_ACTIVE_WRITER_BIT, 0, "simultaneously active INTENT and exclusive locks during exclusive lock acquisition");
98            return (false, was_previously_pending);
99        }
100        if prev_state & RWLOCK_STATE_ACTIVE_WRITER_BIT != 0 {
101            debug_assert_eq!(prev_state & RWLOCK_STATE_ACTIVE_INTENT_BIT, 0, "simultaneously active intent and EXCLUSIVE locks during exclusive lock acquisition");
102            return (false, was_previously_pending);
103        }
104
105        let success = self
106            .state
107            .compare_exchange(
108                current_state,
109                (current_state + 1) | RWLOCK_STATE_ACTIVE_WRITER_BIT,
110                // make sure that the lock is updated to be an exclusive lock, before any
111                // subsequent operations assume that underlying data to be readable (success
112                // ordering).
113                Ordering::Acquire,
114                // a failed attempt to attain an exclusive lock means that the lock cannot later be
115                // used for data accesses, and thus no synchronization is needed here (failure
116                // ordering).
117                Ordering::Relaxed,
118            )
119            .is_ok();
120        (success, was_previously_pending)
121    }
122    unsafe fn try_upgrade_raw(&self) -> (bool, bool) {
123        let prev = self
124            .state
125            .fetch_or(RWLOCK_STATE_PENDING_WRITER_BIT, Ordering::Relaxed);
126
127        debug_assert_ne!(
128            prev & RWLOCK_STATE_ACTIVE_INTENT_BIT,
129            0,
130            "upgrading an intent lock into an exclusive lock when no intent lock was held"
131        );
132
133        let was_previously_pending = prev & RWLOCK_STATE_PENDING_WRITER_BIT != 0;
134
135        let prev = self.state.compare_exchange_weak(
136            RWLOCK_STATE_ACTIVE_INTENT_BIT | RWLOCK_STATE_PENDING_WRITER_BIT | 1,
137            RWLOCK_STATE_ACTIVE_WRITER_BIT | 1,
138            // make sure that the no operations that rely on the lock being an exclusive lock,
139            // happen *after* the lock has been marked exclusive (success ordering).
140            Ordering::Acquire,
141            // no synchronization needs to happen, since a lock is neither obtained nor released
142            // (failure ordering).
143            Ordering::Relaxed,
144        );
145
146        if let Ok(prev_raw) = prev {
147            debug_assert_eq!(
148                prev_raw & RWLOCK_STATE_ACTIVE_WRITER_BIT,
149                0,
150                "upgrading an intent lock into an exclusive lock when an exclusive lock was held"
151            );
152        }
153
154        (prev.is_ok(), was_previously_pending)
155    }
156}

158unsafe impl lock_api::RawRwLock for RawRwLock {
159    const INIT: Self = RawRwLock::const_new();
160
161    type GuardMarker = lock_api::GuardNoSend;
162
163    fn lock_shared(&self) {
164        while !self.try_lock_shared() {
165            atomic::spin_loop_hint();
166        }
167    }
168    fn try_lock_shared(&self) -> bool {
169        let prev = self.state.fetch_add(1, Ordering::Acquire);
170
171        if prev & RWLOCK_STATE_PENDING_WRITER_BIT != 0 {
172            // don't starve writers; writers are prioritized over readers
173            return false;
174        }
175
176        if prev & RWLOCK_STATE_ACTIVE_WRITER_BIT != 0 {
177            let new_prev = self.state.fetch_sub(1, Ordering::Release);
178            debug_assert_ne!(
179                new_prev & !(RWLOCK_STATE_ACTIVE_WRITER_BIT | RWLOCK_STATE_ACTIVE_INTENT_BIT),
180                0,
181                "overflow when subtracting rwlock counter"
182            );
183            return false;
184        }
185        true
186    }
187    fn lock_exclusive(&self) {
188        while !self.try_lock_exclusive() {
189            atomic::spin_loop_hint();
190        }
191    }
192    fn try_lock_exclusive(&self) -> bool {
193        let (success, was_previously_pending) = self.try_lock_exclusive_raw();
194
195        if success && !was_previously_pending {
196            self.state
197                .fetch_and(!RWLOCK_STATE_PENDING_WRITER_BIT, Ordering::Relaxed);
198        }
199
200        success
201    }
202
203    // releases a shared lock
204    unsafe fn unlock_shared(&self) {
205        let prev = self.state.fetch_sub(1, Ordering::Release);
206        debug_assert_ne!(
207            prev & RWLOCK_STATE_COUNT_MASK,
208            0,
209            "corrupted state flags because of subtraction overflow, when release a shared lock"
210        );
211        debug_assert_eq!(
212            prev & RWLOCK_STATE_ACTIVE_WRITER_BIT,
213            0,
214            "releasing a shared lock while an exclusive lock was held"
215        );
216    }
217    // releases an exclusive lock
218    unsafe fn unlock_exclusive(&self) {
219        let prev = self
220            .state
221            .fetch_sub(RWLOCK_STATE_ACTIVE_WRITER_BIT | 1, Ordering::Release);
222        debug_assert_ne!(prev & RWLOCK_STATE_ACTIVE_WRITER_BIT, 0, "corrupted state flags because an exclusive lock release was tried when an exclusive lock was not held");
223        debug_assert_eq!(
224            prev & RWLOCK_STATE_ACTIVE_INTENT_BIT,
225            0,
226            "releasing an exclusive lock when an intent lock was held"
227        );
228    }
229}
230unsafe impl lock_api::RawRwLockDowngrade for RawRwLock {
231    // downgrades an exclusive lock to a shared lock
232    unsafe fn downgrade(&self) {
233        let prev = self
234            .state
235            .fetch_and(!RWLOCK_STATE_ACTIVE_WRITER_BIT, Ordering::Release);
236        debug_assert_ne!(
237            prev & RWLOCK_STATE_ACTIVE_WRITER_BIT,
238            0,
239            "downgrading an exclusive lock to a shared lock when no exclusive lock was held"
240        );
241        debug_assert_eq!(
242            prev & RWLOCK_STATE_ACTIVE_INTENT_BIT,
243            0,
244            "downgrading a exclusive lock to a shared lock when an intent lock was held"
245        );
246    }
247}
248unsafe impl lock_api::RawRwLockUpgrade for RawRwLock {
249    // acquires an intent lock
250    fn lock_upgradable(&self) {
251        while !self.try_lock_upgradable() {
252            atomic::spin_loop_hint();
253        }
254    }
255    // tries to acquire an intent lock
256    fn try_lock_upgradable(&self) -> bool {
257        use lock_api::RawRwLock as _;
258
259        // Begin by acquiring a shared lock.
260        if !self.try_lock_shared() {
261            return false;
262        };
263
264        // At this stage we know that it is completely impossible for a exclusive lock to exist, since
265        // try_lock_shared() would return false in that case. Hence, all we have to do is setting
266        // the active intent bit, and returning false if it was already set.
267        let prev = self
268            .state
269            .fetch_or(RWLOCK_STATE_ACTIVE_INTENT_BIT, Ordering::AcqRel);
270        debug_assert_eq!(
271            prev & RWLOCK_STATE_ACTIVE_WRITER_BIT,
272            0,
273            "acquiring an intent lock while an exclusive lock was held"
274        );
275
276        prev & RWLOCK_STATE_ACTIVE_INTENT_BIT == 0
277    }
278    // releases an intent lock
279    unsafe fn unlock_upgradable(&self) {
280        // assumes that the lock is properly managed by lock_api; if RWLOCK_STATE_ACTIVE_INTENT_BIT
281        // is not set and this method is called, the CPU will arithmetically borrow the bits below,
282        // potentially corrupting the rwlock state entirely.
283        let prev = self
284            .state
285            .fetch_sub(RWLOCK_STATE_ACTIVE_INTENT_BIT | 1, Ordering::Release);
286        debug_assert_ne!(
287            prev & RWLOCK_STATE_ACTIVE_INTENT_BIT,
288            0,
289            "releasing an intent lock while no intent lock was held"
290        );
291        debug_assert_eq!(
292            prev & RWLOCK_STATE_ACTIVE_WRITER_BIT,
293            0,
294            "releasing an intent lock while an exclusive lock was held"
295        );
296    }
297    // upgrades an intent lock into an exclusive lock
298    unsafe fn upgrade(&self) {
299        while !self.try_upgrade() {
300            atomic::spin_loop_hint();
301        }
302    }
303
304    // tries to upgrade an intent lock into an exclusive lock
305    unsafe fn try_upgrade(&self) -> bool {
306        let (success, was_previously_pending) = self.try_upgrade_raw();
307
308        if success && !was_previously_pending {
309            self.state
310                .fetch_and(!RWLOCK_STATE_PENDING_WRITER_BIT, Ordering::Relaxed);
311        }
312
313        success
314    }
315}
316unsafe impl lock_api::RawRwLockUpgradeDowngrade for RawRwLock {
317    // downgrades an exclusive lock to an intent lock
318    unsafe fn downgrade_to_upgradable(&self) {
319        let prev = self.state.fetch_xor(
320            RWLOCK_STATE_ACTIVE_WRITER_BIT | RWLOCK_STATE_ACTIVE_INTENT_BIT,
321            // ensure that previous memory accesses, which may have been writes (as we are
322            // downgrading an exclusive lock), happen before other threads see this downgrade.
323            Ordering::Release,
324        );
325        debug_assert_ne!(
326            prev & RWLOCK_STATE_ACTIVE_WRITER_BIT,
327            0,
328            "downgrading a exclusive lock to an intent lock when no exclusive lock was held"
329        );
330        debug_assert_eq!(
331            prev & RWLOCK_STATE_ACTIVE_INTENT_BIT,
332            0,
333            "downgrading a exclusive lock to an intent lock when an intent lock was held"
334        );
335    }
336    // downgrades an intent lock into a shared lock
337    unsafe fn downgrade_upgradable(&self) {
338        let prev = self.state.fetch_and(
339            !RWLOCK_STATE_ACTIVE_INTENT_BIT,
340            // this ordering is correct, because no memory accesses that may have happened
341            // before, or are going to happen afterwards, will have different rules regarding
342            // mutability (intent locks are only shared locks but with the exception that there
343            // can only be one intent lock at a time).
344            Ordering::Relaxed,
345        );
346        debug_assert_eq!(
347            prev & RWLOCK_STATE_ACTIVE_WRITER_BIT,
348            0,
349            "downgrading an intent lock while a exclusive lock was held"
350        );
351        debug_assert_ne!(
352            prev & RWLOCK_STATE_ACTIVE_INTENT_BIT,
353            0,
354            "downgrading an intent lock where no intent lock was held"
355        );
356    }
357}

371/// A synchronization primitive which initializes a value lazily, once. Since this also includes a
372/// value, it is a bit more like `Once` from `parking_lot` or `spin`.
373#[derive(Debug)]
374pub struct Once<T> {
375    state: AtomicU8,
376    value: UnsafeCell<MaybeUninit<T>>,
377}
378
379impl<T> Drop for Once<T> {
380    fn drop(&mut self) {
381        // we do not have to do any complex state manipulation here, since a mutable reference
382        // guarantees that only there is an exclusive borrow to this struct.
383        if *self.state.get_mut() != OnceState::Initialized as u8 {
384            // nothing to drop
385            return;
386        }
387        unsafe { ptr::drop_in_place(self.value.get() as *mut T) }
388    }
389}
390
391unsafe impl<T: Send + Sync> Send for Once<T> {}
392unsafe impl<T: Send + Sync> Sync for Once<T> {}

394#[repr(u8)]
395#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
396pub enum OnceState {
397    Uninitialized = 0,
398    Initializing = 1,
399    Initialized = 2,
400}
401
402impl<T> Once<T> {
403    pub const fn new() -> Self {
404        Self {
405            state: AtomicU8::new(OnceState::Uninitialized as u8),
406            value: UnsafeCell::new(MaybeUninit::uninit()),
407        }
408    }
409    pub const fn uninitialized() -> Self {
410        Self::new()
411    }
412    pub const fn initialized(value: T) -> Self {
413        Self {
414            state: AtomicU8::new(OnceState::Initialized as u8),
415            value: UnsafeCell::new(MaybeUninit::new(value)),
416        }
417    }
418    pub fn initialize(&self, value: T) -> Result<(), T> {
419        match self.state.compare_exchange(
420            OnceState::Uninitialized as u8,
421            OnceState::Initializing as u8,
422            // make sure that the actual pointer store must happen after other threads have seen
423            // the updated state (success ordering).
424            Ordering::Acquire,
425            // if the once was not uninitialized, there will not be any additional stores, and
426            // since we just return upon failure, Relaxed suffices here.
427            Ordering::Relaxed,
428        ) {
429            Ok(_) => {
430                unsafe {
431                    (*self.value.get()).as_mut_ptr().write(value);
432                };
433                let old = self
434                    .state
435                    .swap(OnceState::Initialized as u8, Ordering::Release);
436                debug_assert_eq!(
437                    old,
438                    OnceState::Initializing as u8,
439                    "once state was modified when setting state to \"initialized\""
440                );
441                Ok(())
442            }
443            Err(_) => Err(value),
444        }
445    }
446    pub fn try_call_once<'a, F>(&'a self, init: F) -> Result<&'a T, F>
447    where
448        F: FnOnce() -> T,
449    {
450        match self.state.compare_exchange(
451            OnceState::Uninitialized as u8,
452            OnceState::Initializing as u8,
453            Ordering::Acquire,
454            Ordering::Relaxed,
455        ) {
456            Ok(_) => unsafe {
457                (*self.value.get()).as_mut_ptr().write(init());
458                let old = self
459                    .state
460                    .swap(OnceState::Initialized as u8, Ordering::Release);
461                debug_assert_eq!(
462                    old,
463                    OnceState::Initializing as u8,
464                    "once state was modified when setting state to \"initialized\""
465                );
466                Ok(&*((*self.value.get()).as_ptr()))
467            },
468            Err(other_state) if other_state == OnceState::Initialized as u8 => unsafe {
469                Ok(&*((*self.value.get()).as_ptr()))
470            },
471
472            #[cfg(debug_assertions)]
473            Err(other_state) if other_state == OnceState::Initializing as u8 => Err(init),
474
475            #[cfg(debug_assertions)]
476            Err(_) => unreachable!(),
477
478            #[cfg(not(debug_assertions))]
479            Err(_) => Err(init),
480        }
481    }
482    pub fn call_once<'a, F>(&'a self, mut init: F) -> &'a T
483    where
484        F: FnOnce() -> T,
485    {
486        loop {
487            match self.try_call_once(init) {
488                Ok(reference) => return reference,
489                Err(init_again) => {
490                    init = init_again;
491                    continue;
492                }
493            }
494        }
495    }
496    pub fn wait<'a>(&'a self) -> &'a T {
497        loop {
498            match self.try_get() {
499                Some(t) => return t,
500                None => continue,
501            }
502        }
503    }
504    pub fn try_get<'a>(&'a self) -> Option<&'a T> {
505        let state = self.state.load(Ordering::Acquire);
506
507        if state != OnceState::Initialized as u8 {
508            return None;
509        }
510        Some(unsafe { &*(self.value.get() as *const T) })
511    }
512    pub fn state(&self) -> OnceState {
513        match self.state.load(Ordering::Acquire) {
514            0 => OnceState::Uninitialized,
515            1 => OnceState::Initializing,
516            2 => OnceState::Initialized,
517            _ => unreachable!(),
518        }
519    }
520}

527/// A value which is initialized on the first access.
528///
529/// This type is thread-safe and can be used in statics.
530///
531/// # Example
532///
533/// ```
534/// use std::collections::HashMap;
535///
536/// use spinning::Lazy;
537///
538/// static HASHMAP: Lazy<HashMap<i32, String>> = Lazy::new(|| {
539///     println!("initializing");
540///     let mut m = HashMap::new();
541///     m.insert(13, "Spica".to_string());
542///     m.insert(74, "Hoyten".to_string());
543///     m
544/// });
545///
546/// fn main() {
547///     println!("ready");
548///     std::thread::spawn(|| {
549///         println!("{:?}", HASHMAP.get(&13));
550///     }).join().unwrap();
551///     println!("{:?}", HASHMAP.get(&74));
552///
553///     // Prints:
554///     //   ready
555///     //   initializing
556///     //   Some("Spica")
557///     //   Some("Hoyten")
558/// }
559/// ```
560pub struct Lazy<T, F = fn() -> T> {
561    once: Once<T>,
562    init: Cell<Option<F>>,
563}
564
565impl<T: fmt::Debug, F> fmt::Debug for Lazy<T, F> {
566    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
567        f.debug_struct("Lazy")
568            .field("once", &self.once)
569            .field("init", &"..")
570            .finish()
571    }
572}

599impl<T, F: FnOnce() -> T> Lazy<T, F> {
600    /// Forces the evaluation of this lazy value and
601    /// returns a reference to the result. This is equivalent
602    /// to the `Deref` impl, but is explicit.
603    ///
604    /// # Example
605    /// ```
606    /// use spinning::Lazy;
607    ///
608    /// let lazy = Lazy::new(|| 92);
609    ///
610    /// assert_eq!(Lazy::force(&lazy), &92);
611    /// assert_eq!(&*lazy, &92);
612    /// ```
613    pub fn force(this: &Lazy<T, F>) -> &T {
614        this.once.call_once(|| match this.init.take() {
615            Some(f) => f(),
616            None => panic!("Lazy instance has previously been poisoned"),
617        })
618    }
619}

623    fn deref(&self) -> &T {
624        Lazy::force(self)
625    }
626}

697    fn singlethread_rwlock() {
698        let data = RwLock::new(1);
699
700        let intent_lock = data.upgradable_read();
701        {
702            let lock1 = data.read();
703            let lock2 = data.read();
704            let lock3 = data.read();
705
706            assert_eq!(*lock1, 1);
707            assert_eq!(*lock2, 1);
708            assert_eq!(*lock3, 1);
709            assert_eq!(*intent_lock, 1);
710        }
711        let mut write_lock = RwLockUpgradableReadGuard::upgrade(intent_lock);
712        *write_lock = 2;
713
714        let intent_lock_again = RwLockWriteGuard::downgrade_to_upgradable(write_lock);
715        let lock1 = {
716            let lock1 = data.read();
717            let lock2 = data.read();
718
719            assert_eq!(*intent_lock_again, 2);
720            assert_eq!(*lock1, 2);
721            assert_eq!(*lock2, 2);
722            lock1
723        };
724        assert!(data.try_write().is_none());
725        let lock3 = RwLockUpgradableReadGuard::downgrade(intent_lock_again);
726        assert_eq!(*lock3, 2);
727        assert_eq!(*lock1, 2);
728    }

743                "upgraded intent lock into exclusive lock while there were still readers"

747            assert_eq!(*lock2, 7);
748            assert_eq!(*lock3, 7);
749
750            upgrade_result.err().unwrap()
751        };
752        let mut write_lock = RwLockUpgradableReadGuard::try_upgrade(upgradable).unwrap();
753        *write_lock = 8;
754        assert_eq!(*write_lock, 8);
