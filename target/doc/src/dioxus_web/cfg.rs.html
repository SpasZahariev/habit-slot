1///  Configuration for the WebSys renderer for the Dioxus VirtualDOM.
2///
3/// This struct helps configure the specifics of hydration and render destination for WebSys.
4///
5/// # Example
6///
7/// ```rust, ignore
8/// dioxus_web::launch(App, Config::new().hydrate(true).root_name("myroot"))
9/// ```
10pub struct Config {
11    pub(crate) hydrate: bool,
12    pub(crate) root: ConfigRoot,
13    pub(crate) default_panic_hook: bool,
14}
15
16pub(crate) enum ConfigRoot {
17    RootName(String),
18    RootElement(web_sys::Element),
19}
20
21impl Config {
22    /// Create a new Default instance of the Config.
23    ///
24    /// This is no different than calling `Config::default()`
25    pub fn new() -> Self {
26        Self::default()
27    }
28
29    #[cfg(feature = "hydrate")]
30    /// Enable SSR hydration
31    ///
32    /// This enables Dioxus to pick up work from a pre-renderd HTML file. Hydration will completely skip over any async
33    /// work and suspended nodes.
34    ///
35    /// Dioxus will load up all the elements with the `dio_el` data attribute into memory when the page is loaded.
36    pub fn hydrate(mut self, f: bool) -> Self {
37        self.hydrate = f;
38        self
39    }
40
41    /// Set the name of the element that Dioxus will use as the root.
42    ///
43    /// This is akin to calling React.render() on the element with the specified name.
44    /// Note that this only works on the current document, i.e. `window.document`.
45    /// To use a different document (popup, iframe, ...) use [Self::rootelement] instead.
46    pub fn rootname(mut self, name: impl Into<String>) -> Self {
47        self.root = ConfigRoot::RootName(name.into());
48        self
49    }
50
51    /// Set the element that Dioxus will use as root.
52    ///
53    /// This is akin to calling React.render() on the given element.
54    pub fn rootelement(mut self, elem: web_sys::Element) -> Self {
55        self.root = ConfigRoot::RootElement(elem);
56        self
57    }
58
59    /// Set whether or not Dioxus should use the built-in panic hook or defer to your own.
60    ///
61    /// The panic hook is set to true normally so even the simplest apps have helpful error messages.
62    pub fn with_default_panic_hook(mut self, f: bool) -> Self {
63        self.default_panic_hook = f;
64        self
65    }
66}
67
68impl Default for Config {
69    fn default() -> Self {
70        Self {
71            hydrate: false,
72            root: ConfigRoot::RootName("main".to_string()),
73            default_panic_hook: true,
74        }
75    }
76}
