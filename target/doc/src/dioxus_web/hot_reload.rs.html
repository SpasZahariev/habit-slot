3use futures_channel::mpsc::UnboundedReceiver;
4
5use dioxus_core::Template;
6use web_sys::Element;
7
8pub(crate) fn init() -> UnboundedReceiver<Template> {
9    use wasm_bindgen::closure::Closure;
10    use wasm_bindgen::JsCast;
11    use web_sys::{MessageEvent, WebSocket};
12
13    use serde::Deserialize;
14
15    let window = web_sys::window().unwrap();
16
17    let protocol = match window.location().protocol().unwrap() {
18        prot if prot == "https:" => "wss:",
19        _ => "ws:",
20    };
21
22    let url = format!(
23        "{protocol}//{}/_dioxus/hot_reload",
24        window.location().host().unwrap()
25    );
26
27    let ws = WebSocket::new(&url).unwrap();
28
29    let (tx, rx) = futures_channel::mpsc::unbounded();
30
31    // change the rsx when new data is received
32    let cl = Closure::wrap(Box::new(move |e: MessageEvent| {
33        if let Ok(text) = e.data().dyn_into::<js_sys::JsString>() {
34            let string: String = text.into();
35
36            if let Ok(val) = serde_json::from_str::<serde_json::Value>(&string) {
37                // leak the value
38                let val: &'static serde_json::Value = Box::leak(Box::new(val));
39                let template: Template = Template::deserialize(val).unwrap();
40                tx.unbounded_send(template).unwrap();
41            } else {
42                // it might be triggering a reload of assets
43                // invalidate all the stylesheets on the page
44                let links = web_sys::window()
45                    .unwrap()
46                    .document()
47                    .unwrap()
48                    .query_selector_all("link[rel=stylesheet]")
49                    .unwrap();
50
51                let noise = js_sys::Math::random();
52
53                for x in 0..links.length() {
54                    let link: Element = links.get(x).unwrap().unchecked_into();
55                    let href = link.get_attribute("href").unwrap();
56                    _ = link.set_attribute("href", &format!("{}?{}", href, noise));
57                }
58            }
59        }
60    }) as Box<dyn FnMut(MessageEvent)>);
61
62    ws.set_onmessage(Some(cl.as_ref().unchecked_ref()));
63    cl.forget();
64
65    rx
66}
