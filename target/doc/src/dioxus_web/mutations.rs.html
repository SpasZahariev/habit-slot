13    pub(crate) fn create_template_node(&self, v: &TemplateNode) -> web_sys::Node {
14        use TemplateNode::*;
15        match v {
16            Element {
17                tag,
18                namespace,
19                attrs,
20                children,
21                ..
22            } => {
23                let el = match namespace {
24                    Some(ns) => self.document.create_element_ns(Some(ns), tag).unwrap(),
25                    None => self.document.create_element(tag).unwrap(),
26                };
27                for attr in *attrs {
28                    if let TemplateAttribute::Static {
29                        name,
30                        value,
31                        namespace,
32                    } = attr
33                    {
34                        minimal_bindings::setAttributeInner(
35                            el.clone().into(),
36                            name,
37                            JsValue::from_str(value),
38                            *namespace,
39                        );
40                    }
41                }
42                for child in *children {
43                    let _ = el.append_child(&self.create_template_node(child));
44                }
45                el.dyn_into().unwrap()
46            }
47            Text { text } => self.document.create_text_node(text).dyn_into().unwrap(),
48            DynamicText { .. } => self.document.create_text_node("p").dyn_into().unwrap(),
49            Dynamic { .. } => {
50                let el = self.document.create_element("pre").unwrap();
51                let _ = el.toggle_attribute("hidden");
52                el.dyn_into().unwrap()
53            }
54        }
55    }

57    pub fn flush_edits(&mut self) {
58        self.interpreter.flush();
59
60        // Now that we've flushed the edits and the dom nodes exist, we can send the mounted events.
61        #[cfg(feature = "mounted")]
62        self.flush_queued_mounted_events();
63    }
64
65    #[cfg(feature = "mounted")]
66    fn flush_queued_mounted_events(&mut self) {
67        for id in self.queued_mounted_events.drain(..) {
68            let node = self.interpreter.base().get_node(id.0 as u32);
69            if let Some(element) = node.dyn_ref::<web_sys::Element>() {
70                let _ = self.event_channel.unbounded_send(UiEvent {
71                    name: "mounted".to_string(),
72                    bubbles: false,
73                    element: id,
74                    data: PlatformEventData::new(Box::new(element.clone())),
75                });
76            }
77        }
78    }
79
80    #[cfg(feature = "mounted")]
81    pub(crate) fn send_mount_event(&mut self, id: ElementId) {
82        self.queued_mounted_events.push(id);
83    }

87    fn register_template(&mut self, template: Template) {
88        let mut roots = vec![];
89        for root in template.roots {
90            roots.push(self.create_template_node(root))
91        }
92        self.templates
93            .insert(template.name.to_owned(), self.max_template_id);
94        self.interpreter
95            .base()
96            .save_template(roots, self.max_template_id);
97        self.max_template_id += 1
98    }

146    fn set_attribute(
147        &mut self,
148        name: &'static str,
149        ns: Option<&'static str>,
150        value: &AttributeValue,
151        id: ElementId,
152    ) {
153        match value {
154            AttributeValue::Text(txt) => {
155                self.interpreter
156                    .set_attribute(id.0 as u32, name, txt, ns.unwrap_or_default())
157            }
158            AttributeValue::Float(f) => self.interpreter.set_attribute(
159                id.0 as u32,
160                name,
161                &f.to_string(),
162                ns.unwrap_or_default(),
163            ),
164            AttributeValue::Int(n) => self.interpreter.set_attribute(
165                id.0 as u32,
166                name,
167                &n.to_string(),
168                ns.unwrap_or_default(),
169            ),
170            AttributeValue::Bool(b) => self.interpreter.set_attribute(
171                id.0 as u32,
172                name,
173                if *b { "true" } else { "false" },
174                ns.unwrap_or_default(),
175            ),
176            AttributeValue::None => {
177                self.interpreter
178                    .remove_attribute(id.0 as u32, name, ns.unwrap_or_default())
179            }
180            _ => unreachable!(),
181        }
182    }

188    fn create_event_listener(&mut self, name: &'static str, id: ElementId) {
189        // mounted events are fired immediately after the element is mounted.
190        if name == "mounted" {
191            #[cfg(feature = "mounted")]
192            self.send_mount_event(id);
193            return;
194        }
195
196        self.interpreter
197            .new_event_listener(name, id.0 as u32, event_bubbles(name) as u8);
198    }
199
200    fn remove_event_listener(&mut self, name: &'static str, id: ElementId) {
201        if name == "mounted" {
202            return;
203        }
204
205        self.interpreter
206            .remove_event_listener(name, id.0 as u32, event_bubbles(name) as u8);
207    }
