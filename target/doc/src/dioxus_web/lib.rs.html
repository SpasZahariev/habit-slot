5//! Dioxus WebSys
6//!
7//! ## Overview
8//! ------------
9//! This crate implements a renderer of the Dioxus Virtual DOM for the web browser using WebSys. This web render for
10//! Dioxus is one of the more advanced renderers, supporting:
11//! - idle work
12//! - animations
13//! - jank-free rendering
14//! - controlled components
15//! - hydration
16//! - and more.
17//!
18//! The actual implementation is farily thin, with the heavy lifting happening inside the Dioxus Core crate.
19//!
20//! To purview the examples, check of the root Dioxus crate - the examples in this crate are mostly meant to provide
21//! validation of websys-specific features and not the general use of Dioxus.
22

51/// Runs the app as a future that can be scheduled around the main thread.
52///
53/// Polls futures internal to the VirtualDOM, hence the async nature of this function.
54///
55/// # Example
56///
57/// ```ignore, rust
58/// let app_fut = dioxus_web::run_with_props(App, RootProps { name: String::from("foo") });
59/// wasm_bindgen_futures::spawn_local(app_fut);
60/// ```
61pub async fn run(virtual_dom: VirtualDom, web_config: Config) {
62    tracing::info!("Starting up");
63
64    let mut dom = virtual_dom;
65
66    #[cfg(feature = "eval")]
67    dom.in_runtime(eval::init_eval);
68
69    #[cfg(feature = "panic_hook")]
70    if web_config.default_panic_hook {
71        console_error_panic_hook::set_once();
72    }
73
74    #[cfg(all(feature = "hot_reload", debug_assertions))]
75    let mut hotreload_rx = hot_reload::init();
76
77    let (tx, mut rx) = futures_channel::mpsc::unbounded();
78
79    let should_hydrate = web_config.hydrate;
80
81    let mut websys_dom = dom::WebsysDom::new(web_config, tx);
82
83    tracing::info!("rebuilding app");
84
85    if should_hydrate {
86        #[cfg(feature = "hydrate")]
87        {
88            dom.rebuild(&mut crate::rehydrate::OnlyWriteTemplates(&mut websys_dom));
89
90            if let Err(err) = websys_dom.rehydrate(&dom) {
91                tracing::error!("Rehydration failed. {:?}", err);
92                tracing::error!("Rebuild DOM into element from scratch");
93                websys_dom.root.set_text_content(None);
94
95                dom.rebuild(&mut websys_dom);
96
97                websys_dom.flush_edits();
98            }
99        }
100    } else {
101        dom.rebuild(&mut websys_dom);
102
103        websys_dom.flush_edits();
104    }
105
106    // the mutations come back with nothing - we need to actually mount them
107    websys_dom.mount();
108
109    loop {
110        // if virtual dom has nothing, wait for it to have something before requesting idle time
111        // if there is work then this future resolves immediately.
112        let (mut res, template) = {
113            let work = dom.wait_for_work().fuse();
114            pin_mut!(work);
115
116            let mut rx_next = rx.select_next_some();
117
118            #[cfg(all(feature = "hot_reload", debug_assertions))]
119            {
120                let mut hot_reload_next = hotreload_rx.select_next_some();
121                select! {
122                    _ = work => (None, None),
123                    new_template = hot_reload_next => (None, Some(new_template)),
124                    evt = rx_next => (Some(evt), None),
125                }
126            }
127
128            #[cfg(not(all(feature = "hot_reload", debug_assertions)))]
129            select! {
130                _ = work => (None, None),
131                evt = rx_next => (Some(evt), None),
132            }
133        };
134
135        if let Some(template) = template {
136            dom.replace_template(template);
137        }
138
139        // Dequeue all of the events from the channel in send order
140        // todo: we should re-order these if possible
141        while let Some(evt) = res {
142            dom.handle_event(
143                evt.name.as_str(),
144                Rc::new(evt.data),
145                evt.element,
146                evt.bubbles,
147            );
148            res = rx.try_next().transpose().unwrap().ok();
149        }
150
151        // Todo: This is currently disabled because it has a negative impact on response times for events but it could be re-enabled for tasks
152        // Jank free rendering
153        //
154        // 1. wait for the browser to give us "idle" time
155        // 2. During idle time, diff the dom
156        // 3. Stop diffing if the deadline is exceded
157        // 4. Wait for the animation frame to patch the dom
158
159        // wait for the mainthread to schedule us in
160        // let deadline = work_loop.wait_for_idle_time().await;
161
162        // run the virtualdom work phase until the frame deadline is reached
163        dom.render_immediate(&mut websys_dom);
164
165        // wait for the animation frame to fire so we can apply our changes
166        // work_loop.wait_for_raf().await;
167
168        websys_dom.flush_edits();
169    }
170}
