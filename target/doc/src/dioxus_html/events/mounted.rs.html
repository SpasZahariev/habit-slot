11/// An Element that has been rendered and allows reading and modifying information about it.
12///
13/// Different platforms will have different implementations and different levels of support for this trait. Renderers that do not support specific features will return `None` for those queries.
14// we can not use async_trait here because it does not create a trait that is object safe
15pub trait RenderedElementBacking: std::any::Any {
16    /// return self as Any
17    fn as_any(&self) -> &dyn std::any::Any;
18
19    /// Get the bounding rectangle of the element relative to the viewport (this does not include the scroll position)
20    #[allow(clippy::type_complexity)]
21    fn get_client_rect(&self) -> Pin<Box<dyn Future<Output = MountedResult<Rect<f64, f64>>>>> {
22        Box::pin(async { Err(MountedError::NotSupported) })
23    }
24
25    /// Scroll to make the element visible
26    fn scroll_to(
27        &self,
28        _behavior: ScrollBehavior,
29    ) -> Pin<Box<dyn Future<Output = MountedResult<()>>>> {
30        Box::pin(async { Err(MountedError::NotSupported) })
31    }
32
33    /// Set the focus on the element
34    fn set_focus(&self, _focus: bool) -> Pin<Box<dyn Future<Output = MountedResult<()>>>> {
35        Box::pin(async { Err(MountedError::NotSupported) })
36    }
37}
38
39impl RenderedElementBacking for () {
40    fn as_any(&self) -> &dyn std::any::Any {
41        self
42    }
43}
44
45/// The way that scrolling should be performed
46#[cfg_attr(feature = "serialize", derive(serde::Serialize, serde::Deserialize))]
47pub enum ScrollBehavior {
48    /// Scroll to the element immediately
49    #[cfg_attr(feature = "serialize", serde(rename = "instant"))]
50    Instant,
51    /// Scroll to the element smoothly
52    #[cfg_attr(feature = "serialize", serde(rename = "smooth"))]
53    Smooth,
54}

56/// An Element that has been rendered and allows reading and modifying information about it.
57///
58/// Different platforms will have different implementations and different levels of support for this trait. Renderers that do not support specific features will return `None` for those queries.
59pub struct MountedData {
60    inner: Box<dyn RenderedElementBacking>,
61}
62
63impl<E: RenderedElementBacking> From<E> for MountedData {
64    fn from(e: E) -> Self {
65        Self { inner: Box::new(e) }
66    }
67}
68
69impl MountedData {
70    /// Create a new MountedData
71    pub fn new(registry: impl RenderedElementBacking + 'static) -> Self {
72        Self {
73            inner: Box::new(registry),
74        }
75    }
76
77    /// Get the bounding rectangle of the element relative to the viewport (this does not include the scroll position)
78    pub async fn get_client_rect(&self) -> MountedResult<Rect<f64, f64>> {
79        self.inner.get_client_rect().await
80    }
81
82    /// Scroll to make the element visible
83    pub fn scroll_to(
84        &self,
85        behavior: ScrollBehavior,
86    ) -> Pin<Box<dyn Future<Output = MountedResult<()>>>> {
87        self.inner.scroll_to(behavior)
88    }
89
90    /// Set the focus on the element
91    pub fn set_focus(&self, focus: bool) -> Pin<Box<dyn Future<Output = MountedResult<()>>>> {
92        self.inner.set_focus(focus)
93    }
94
95    /// Downcast this event to a concrete event type
96    pub fn downcast<T: 'static>(&self) -> Option<&T> {
97        self.inner.as_any().downcast_ref::<T>()
98    }
99}

112/// The MountedResult type for the MountedData
113pub type MountedResult<T> = Result<T, MountedError>;
114
115#[derive(Debug)]
116/// The error type for the MountedData
117#[non_exhaustive]
118pub enum MountedError {
119    /// The renderer does not support the requested operation
120    NotSupported,
121    /// The element was not found
122    OperationFailed(Box<dyn std::error::Error>),
123}
124
125impl Display for MountedError {
126    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
127        match self {
128            MountedError::NotSupported => {
129                write!(f, "The renderer does not support the requested operation")
130            }
131            MountedError::OperationFailed(e) => {
132                write!(f, "The operation failed: {}", e)
133            }
134        }
135    }
136}
